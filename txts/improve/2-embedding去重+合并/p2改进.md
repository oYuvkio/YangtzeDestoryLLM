### 6. 缺少增量去重的支持

当前每次都是全量去重，建议支持增量模式（只处理新增的元素）。

**改进**：

```python
def dedup_incremental(
    new_elements: List[Dict[str, Any]],
    existing_elements: List[Dict[str, Any]],
    model_name: str,
    sim_th: float,
    elem_type: str,
) -> Tuple[List[Dict[str, Any]], List[Dict[str, Any]]]:
    """增量去重：只检查新元素是否与已有元素重复。
  
    Returns:
        (accepted_new, rejected_new)
    """
    if not new_elements:
        return [], []
  
    dedup = EmbeddingDeduplicator(model_name=model_name, threshold=sim_th)
  
    # 预计算已有元素的 embedding
    existing_texts = [_elem_to_text(e) for e in existing_elements]
    existing_embeddings = dedup.encode(existing_texts) if existing_texts else None
  
    accepted = []
    rejected = []
  
    for elem in new_elements:
        elem_text = _elem_to_text(elem)
        elem_embedding = dedup.encode([elem_text])[0]
      
        # 与已有元素比较
        max_sim = 0.0
        best_match = None
      
        if existing_embeddings is not None:
            for i, existing_emb in enumerate(existing_embeddings):
                sim = cosine_similarity(elem_embedding, existing_emb)
                if sim > max_sim:
                    max_sim = sim
                    best_match = existing_elements[i]
      
        # 与已接受的新元素比较
        for acc_elem in accepted:
            acc_text = _elem_to_text(acc_elem)
            acc_emb = dedup.encode([acc_text])[0]
            sim = cosine_similarity(elem_embedding, acc_emb)
            if sim > max_sim:
                max_sim = sim
                best_match = acc_elem
      
        if max_sim >= sim_th:
            elem["_similarity"] = max_sim
            elem["_similar_to"] = best_match.get("name", "") if best_match else ""
            rejected.append(elem)
        else:
            accepted.append(elem)
  
    return accepted, rejected
```
