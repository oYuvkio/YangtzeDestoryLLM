    split_cfg = cfg.get("corpus_split", {})

    if split_cfg.get("p4_keywords"):

    def __init__(
        self,
        manifest_path: Path,

        split_level: str = "doc",
        fold_filter: Optional[str] = None,  # 新增：仅导出特定 fold

        self.fold_filter = fold_filter
        # ... 其他初始化 ...
  
    def should_include(self, doc: Dict[str, Any]) -> bool:
        """判断文档是否应该被包含"""
        doc_id = self._get_doc_id(doc)
        entry = self.manifest.get(doc_id)
      
        if not entry:
            return False
      
        # 检查用途
        if entry.get("purpose") not in self.purposes:
            return False
      
        # 检查 fold（如果指定）
        if self.fold_filter and entry.get("fold") != self.fold_filter:
            return False
